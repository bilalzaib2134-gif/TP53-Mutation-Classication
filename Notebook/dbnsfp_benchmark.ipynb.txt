    "## Step 1 \u2014 Identify the true missense subset\n",
    "\n",
    "Uses ClinVar's native `Molecular consequence` field with a case-insensitive substring match, applied directly to the model's actual test partition.\n",
    "\n",
    "- **Substring match, not exact match:** correctly retains variants with multi-term consequence annotations (e.g., `\"missense_variant|splice_region_variant\"`), since `\"missense\"` appears anywhere in the string.\n",
    "- **`na=False`:** excludes any variant with a missing/absent consequence annotation from the missense subset, rather than silently including it."
   ]

    "def build_hgvs(spdi):\n",
    "    \"\"\"Convert a ClinVar Canonical SPDI string to a 1-based HGVS genomic identifier.\n",
    "    SPDI position is 0-based; HGVS genomic notation is 1-based.\n",
    "    e.g. 'NC_000017.11:7674892:C:G' -> 'chr17:g.7674893C>G'\n",
    "    \"\"\"\n",
    "    parts = spdi.split(':')\n",
    "    pos_1based = int(parts[1]) + 1\n",
    "    return f\"chr17:g.{pos_1based}{parts[2]}>{parts[3]}\"\n",
    "\n",
    "# Pull annotation columns for the model's actual test-set rows only\n",
    "test_consequence = df_raw_filtered.loc[test_df.index, ['Canonical SPDI', 'Molecular consequence', 'Protein change']].copy()\n",
    "test_consequence['label'] = test_df['label'].values\n",
    "\n",
    "missense_mask = test_consequence['Molecular consequence'].str.contains('missense', case=False, na=False)\n",
    "missense_subset = test_consequence[missense_mask].copy()\n",
    "missense_subset['hgvs'] = missense_subset['Canonical SPDI'].apply(build_hgvs)\n",
    "\n",
    "print(f\"Missense variants in test set: {len(missense_subset)}\")\n",
    "print(f\"Pathogenic: {(missense_subset['label']==1).sum()} | Benign: {(missense_subset['label']==0).sum()}\")\n",
    "# Expected: 75 variants (45 pathogenic, 30 benign)"

    "## Step 2 \u2014 Query dbNSFP v4 via the myvariant.info API\n",
    "\n",
    "One live HTTP request per variant. `assembly='hg38'` is required \u2014 the ClinVar coordinates used throughout this project are GRCh38, and myvariant.info defaults to GRCh37/hg19 if not told otherwise; omitting this parameter silently returns no matches rather than raising an error.\n",
    "\n",
    "Both categorical predictions and continuous scores are requested:\n",
    "- **SIFT / PolyPhen-2:** categorical predictions only support sensitivity/specificity. Continuous scores are also pulled to compute AUC. Note SIFT's score is inverted (lower = more damaging) relative to PolyPhen-2 and ClinPred (higher = more damaging) \u2014 handled at analysis time, not here.\n",
    "- **ClinPred:** provides a continuous, calibrated score directly, supporting AUC without any sign flip.\n",
    "- **Worst-case transcript rule:** TP53 has multiple annotated transcript isoforms with non-aligned per-transcript scores in dbNSFP. The most damaging prediction/score across all annotated transcripts is used throughout, consistent with the conservative convention described in the manuscript's Methods section. This likely inflates the apparent false-positive rate relative to canonical-transcript-only scoring."
   ]

   "source": [
    "import requests\n",
    "import time\n",
    "import pandas as pd\n",
    "\n",
    "def worst_sift(pred_list):\n",
    "    \"\"\"SIFT worst-case across transcripts: 'D' (damaging) if any transcript calls it damaging.\"\"\"\n",
    "    if not pred_list:\n",
    "        return None\n",
    "    return 'D' if 'D' in pred_list else 'T'\n",
    "\n",
    "def worst_polyphen(pred_list):\n",
    "    \"\"\"PolyPhen-2 worst-case across transcripts: most damaging call takes precedence (D > P > B).\"\"\"\n",
    "    if not pred_list:\n",
    "        return None\n",
    "    if 'D' in pred_list:\n",
    "        return 'D'\n",
    "    if 'P' in pred_list:\n",
    "        return 'P'\n",
    "    return 'B'\n",
    "\n",
    "fields = (\"dbnsfp.sift.pred,dbnsfp.polyphen2.hdiv.pred,\"\n",
    "          \"dbnsfp.clinpred.pred,dbnsfp.clinpred.score,\"\n",
    "          \"dbnsfp.sift.score,dbnsfp.polyphen2.hdiv.score\")\n",
    "\n",
    "records = []\n",
    "missing = []\n",
    "\n",
    "for idx, row in missense_subset.iterrows():\n",
    "    hgvs = row['hgvs']\n",
    "    url = f\"https://myvariant.info/v1/variant/{hgvs}\"\n",
    "    try:\n",
    "        r = requests.get(url, params={\"fields\": fields, \"assembly\": \"hg38\"}, timeout=10)\n",
    "        data = r.json()\n",
    "        dbnsfp = data.get('dbnsfp', {})\n",
    "\n",
    "        sift_preds = dbnsfp.get('sift', {}).get('pred', [])\n",
    "        if isinstance(sift_preds, str):\n",
    "            sift_preds = [sift_preds]\n",
    "\n",
    "        polyphen_preds = dbnsfp.get('polyphen2', {}).get('hdiv', {}).get('pred', [])\n",
    "        if isinstance(polyphen_preds, str):\n",
    "            polyphen_preds = [polyphen_preds]\n",
    "\n",
    "        clinpred = dbnsfp.get('clinpred', {})\n",
    "\n",
    "        sift_score = dbnsfp.get('sift', {}).get('score', None)\n",
    "        if isinstance(sift_score, list):\n",
    "            sift_score = min(sift_score)  # lower = more damaging -> worst case is minimum\n",
    "\n",
    "        pp_score = dbnsfp.get('polyphen2', {}).get('hdiv', {}).get('score', None)\n",
    "        if isinstance(pp_score, list):\n",
    "            pp_score = max(pp_score)  # higher = more damaging -> worst case is maximum\n",
    "\n",
    "        records.append({\n",
    "            'hgvs': hgvs,\n",
    "            'label': row['label'],\n",
    "            'sift_pred': worst_sift(sift_preds),\n",
    "            'polyphen_pred': worst_polyphen(polyphen_preds),\n",
    "            'clinpred_pred': clinpred.get('pred'),\n",
    "            'clinpred_score': clinpred.get('score'),\n",
    "            'sift_score': sift_score,\n",
    "            'pp_score': pp_score,\n",
    "            'has_dbnsfp': bool(dbnsfp)\n",
    "        })\n",
    "        if not dbnsfp:\n",
    "            missing.append(hgvs)\n",
    "\n",
    "    except Exception as e:\n",
    "        records.append({'hgvs': hgvs, 'label': row['label'], 'error': str(e)})\n",
    "        missing.append(hgvs)\n",
    "\n",
    "    time.sleep(0.1)  # be polite to the free API\n",
    "\n",
    "results_df = pd.DataFrame(records)\n",
    "print(f\"Total queried: {len(results_df)}\")\n",
    "print(f\"Missing/no dbNSFP data: {len(missing)} {missing}\")\n",
    "print()\n",
    "print(results_df['sift_pred'].value_counts(dropna=False))\n",
    "print()\n",
    "print(results_df['polyphen_pred'].value_counts(dropna=False))\n",
    "print()\n",
    "print(results_df['clinpred_pred'].value_counts(dropna=False))"
   ]

    "## Step 3 \u2014 Compute metrics\n",
    "\n",
    "Sensitivity/specificity for all three tools at their native categorical thresholds, plus AUC-ROC wherever a continuous score is available (ClinPred always; SIFT and PolyPhen-2 via their underlying dbNSFP scores, sign-corrected as needed).\n",
    "\n",
    "PolyPhen-2 is reported under both conventions used in the literature: strict (`D` only = pathogenic) and broad (`D`+`P` = pathogenic)."

    "from sklearn.metrics import roc_auc_score, confusion_matrix\n",
    "\n",
    "# ---- ClinPred: continuous score, direct AUC ----\n",
    "clinpred_df = results_df.dropna(subset=['clinpred_score'])\n",
    "y_true_cp = clinpred_df['label'].values\n",
    "y_score_cp = clinpred_df['clinpred_score'].values\n",
    "auc_cp = roc_auc_score(y_true_cp, y_score_cp)\n",
    "y_pred_cp = (clinpred_df['clinpred_pred'] == 'D').astype(int)\n",
    "tn_cp, fp_cp, fn_cp, tp_cp = confusion_matrix(y_true_cp, y_pred_cp).ravel()\n",
    "print(f\"ClinPred: n={len(clinpred_df)}, AUC={auc_cp:.4f}, \"\n",
    "      f\"Sens={tp_cp/(tp_cp+fn_cp):.4f}, Spec={tn_cp/(tn_cp+fp_cp):.4f} \"\n",
    "      f\"(TN={tn_cp} FP={fp_cp} FN={fn_cp} TP={tp_cp})\")\n",
    "print()\n",
    "\n",
    "# ---- SIFT: categorical + continuous (score sign-flipped: lower = more damaging) ----\n",
    "sift_df = results_df.dropna(subset=['sift_pred'])\n",
    "y_true_s = sift_df['label'].values\n",
    "y_pred_s = (sift_df['sift_pred'] == 'D').astype(int)\n",
    "tn_s, fp_s, fn_s, tp_s = confusion_matrix(y_true_s, y_pred_s).ravel()\n",
    "print(f\"SIFT: n={len(sift_df)}, Sens={tp_s/(tp_s+fn_s):.4f}, Spec={tn_s/(tn_s+fp_s):.4f} \"\n",
    "      f\"(TN={tn_s} FP={fp_s} FN={fn_s} TP={tp_s})\")\n",
    "\n",
    "sift_auc_df = results_df.dropna(subset=['sift_score'])\n",
    "auc_sift = roc_auc_score(sift_auc_df['label'], -sift_auc_df['sift_score'])  # flip: lower score = more pathogenic\n",

    "print()\n",
    "\n",
    "# ---- PolyPhen-2: categorical (strict + broad) + continuous ----\n",
    "pp_df = results_df.dropna(subset=['polyphen_pred'])\n",

    "\n",
    "y_pred_p_strict = (pp_df['polyphen_pred'] == 'D').astype(int)\n",

    "print(f\"PolyPhen-2 (D only = pathogenic): n={len(pp_df)}, \"\n",
    "      f\"Sens={tp_p1/(tp_p1+fn_p1):.4f}, Spec={tn_p1/(tn_p1+fp_p1):.4f} \"\n",
    "      f\"(TN={tn_p1} FP={fp_p1} FN={fn_p1} TP={tp_p1})\")\n",
    "\n",
    "y_pred_p_broad = (pp_df['polyphen_pred'].isin(['D', 'P'])).astype(int)\n",
    "tn_p2, fp_p2, fn_p2, tp_p2 = confusion_matrix(y_true_p, y_pred_p_broad).ravel()\n",
    "print(f\"PolyPhen-2 (D+P = pathogenic): n={len(pp_df)}, \"\n",
    "      f\"Sens={tp_p2/(tp_p2+fn_p2):.4f}, Spec={tn_p2/(tn_p2+fp_p2):.4f} \"\n",
    "      f\"(TN={tn_p2} FP={fp_p2} FN={fn_p2} TP={tp_p2})\")\n",
    "\n",
    "pp_auc_df = results_df.dropna(subset=['pp_score'])\n",
    "auc_pp = roc_auc_score(pp_auc_df['label'], pp_auc_df['pp_score'])  # higher score = more pathogenic, no flip needed\n",
    "print(f\"PolyPhen-2 AUC: n={len(pp_auc_df)}, AUC={auc_pp:.4f}\")"
